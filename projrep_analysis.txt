bLineMode = vbFalse

	'define function default value
	Excel_Value = True

	'deactivate error handling
	On Error Resume Next

	'create excel object
	Set oApplication = CreateObject("Excel.Application")
	If Not isObject(oApplication) Then
		Reporter.ReportEvent micFail, "Function Excel_Value - Excel", "Die Anwendung Excel konnte nicht gestartet werden!"
		Excel_Value = False
		Exit Function
	End If

	'set app properties (invisible, non-Interactive)
	oApplication.Visible = False
	oApplication.DisplayAlerts = False

	'open excel file
	Set oWorkBook = oApplication.Workbooks.Open(sFile, , False)
	If Not isObject(oWorkBook) Then
		Reporter.ReportEvent micFail, "Function Excel_Value - Lade Datei", "Die Excel-Datei konnte nicht verarbeitet werden!"
		Excel_Value = False
		Call Excel_DestroyApplication(oApplication)
		Exit Function
	End If

	'set default shett to 1 if not empty
	If Len(sWorksheet) = 0 Then
		sWorksheet = "#1"
	End If
	'select and activate sheet
	If CheckNumeric(sWorksheet) Then
		Set oSheet = oWorkBook.Sheets(CInt(Right(sWorksheet, Len(sWorksheet) - 1)))
	Else
		Set oSheet = oWorkBook.Sheets(sWorksheet)
	End If
	If Not isObject(oSheet) Then
		Reporter.ReportEvent micFail, "Function Excel_Value - Aktiviere Tabellenblatt", "Das angeforderte Tabellenblatt konte in der Excel-Datei nicht gefunden werden!"
		Excel_Value = False
		Call Excel_DestroyWorkbook(oWorkBook)
		Call Excel_Des
